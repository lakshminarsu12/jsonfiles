mediaRecorder.ondataavailable = function (event) {
    if (event.data && event.data.size > 0) {
        fullChunks.push(event.data);

        while (fullChunks.length > MAX_VIDEO_CHUNKS) {
            fullChunks.shift();
        }

        callLogInfo(
            'chunk added. chunks=' +
            fullChunks.length +
            ', chunkSize=' +
            event.data.size
        );
    }
};

mediaRecorder.start(RECORDER_TIMESLICE_MS);

function getFullVideoBlob() {
    if (!fullChunks.length) {
        callLogInfo('No video chunks available');
        return null;
    }

    const blob = new Blob(fullChunks, {
        type: 'video/webm'
    });

    callLogInfo(
        'Created latest video blob. chunks=' +
        fullChunks.length +
        ', size=' +
        blob.size
    );

    return blob;
}

function uploadCurrentVideo(reason, rgIdentifiedFlag) {
    try {
        if (uploadInProgress || processingStopped) {
            callLogInfo(
                'Upload skipped. uploadInProgress=' +
                uploadInProgress +
                ', processingStopped=' +
                processingStopped
            );
            return;
        }

        uploadInProgress = true;

        callLogInfo('uploadCurrentVideo started. reason=' + reason);

        const videoBlob = getFullVideoBlob();

        if (!videoBlob) {
            callLogInfo('uploadCurrentVideo stopped. No video blob.');
            uploadInProgress = false;
            return;
        }

        const ban = getQueryParam("ban") || "";

        const formData = new FormData();
        formData.append('video', videoBlob, 'last5seconds.webm');
        formData.append('ban', ban);
        formData.append('scanTime', scanDuration || (Date.now() - scanStartTime));
        formData.append('blinkSpeed', chossenSpeed || 'uncertain');
        formData.append('blinkColor', choosenColor || 'off');
        formData.append('rgIdentified', rgIdentifiedFlag ? 'true' : 'false');
        formData.append('failureReason', reason || '');

        callLogInfo(
            'calling uploadVideo. size=' +
            videoBlob.size +
            ', rgIdentified=' +
            rgIdentifiedFlag
        );

        fetch('/eapi/1/aiService/uploadVideo', {
            method: 'POST',
            body: formData
        })
        .then(res => {
            callLogInfo('uploadVideo response status=' + res.status);
            return res.text();
        })
        .then(t => {
            callLogInfo('uploadVideo success=' + t);
        })
        .catch(err => {
            callLogInfo('uploadVideo error=' + (err.message || err));
        })
        .finally(() => {
            uploadInProgress = false;
            processingStopped = true;
            callLogInfo('upload finished. processingStopped=true');
        });

    } catch (e) {
        uploadInProgress = false;
        callLogInfo('uploadCurrentVideo exception=' + (e.message || e));
    }
}

function processFrame() {
    if (processingStopped) {
        return;
    }

    document.getElementById("cameraDimensions").textContent =
        `${tfTester.rgConfidence} || ${rgDetector.lastMean.toFixed(1)}`;

    try {
        if (!scanStartTime) {
            scanStartTime = Date.now();
        }

        // RG not confirmed within timeout - upload failed scan only once
        if (
            !rgHandled &&
            !failureUploadHandled &&
            !uploadInProgress &&
            scanStartTime &&
            (Date.now() - scanStartTime > MAX_SCAN_TIME_MS)
        ) {
            failureUploadHandled = true;
            processingStopped = true;

            scanEndTime = Date.now();
            scanDuration = scanEndTime - scanStartTime;

            callLogInfo("RG not confirmed timeout. Uploading video once.");

            uploadCurrentVideo("RG_NOT_CONFIRMED_TIMEOUT", false);

            const msg = JSON.stringify({
                rgIdentified: false,
                rgModel: "UNKNOWN",
                leds: {
                    serviceLed: {
                        color: choosenColor || "off",
                        speed: chossenSpeed || "uncertain"
                    },
                    wpsLed: {
                        color: "off",
                        speed: "uncertain"
                    }
                },
                debug: {
                    failureReason: "RG_NOT_CONFIRMED_TIMEOUT",
                    scanDuration: scanDuration
                }
            });

            try {
                window.ReactNativeWebView.postMessage(msg);
            } catch (e) {}

            try {
                window.ThinNative.postMessage(msg);
            } catch (e) {}

            try {
                window.webkit.messageHandlers.nativeCallBack.postMessage(msg);
            } catch (e) {}

            return;
        }

        if (preview.readyState !== 4) {
            setTimeout(processFrame, 50);
            return;
        }

        hiddenContext.drawImage(preview, 0, 0, hiddenCanvas.width, hiddenCanvas.height);

        (async () => {
            await tfTester.predict(hiddenCanvas);
        })();

        if (!tfTester.rgFound) {

                scanStartTime = Date.now();
            }

            this.overlayImg.src = "ghost2.png";

                detectionResultDiv.textContent = "";
            }

            setTimeout(processFrame, 50);
            return;
        }

        let frame = cv.imread(hiddenCanvas);
        const results = ledDetector.detectLEDColor(frame);
        const rgFound = rgDetector.checkIfRg(frame);

        // Display result text
        choosenColor = null;
        chossenSpeed = null;

        if (results.length > 0) {
            if (results.length === 1) {
                detectionResultDiv.textContent =

                detectionResultDiv.style.color = results[0].color.toLowerCase();

                // Multiple LEDs detected
                const colorList = results.map(r => `${r.color} (${Math.round(r.hue)})`).join(", ");

                    `Detected ${results.length} LEDs: ${colorList}`;
                detectionResultDiv.style.color = "black";
            }
            // Blinking status for the strongest detection
            const r0 = results[0];
            let statusMsg = "";

            if (r0.isSolid) {
                statusMsg = `${r0.color}: steady`;
            } else if (r0.blinkHz && r0.blinkHz > 0) {
                statusMsg = `${r0.color}: blinking ${r0.blinkHz} Hz`;
            } else {
                statusMsg = `${r0.color}: on (estimating…)`;
            }

            blinkingStatusDiv.textContent = statusMsg;
            // Colorize text to the detected color when readable
            const cssColor = r0.color ? r0.color.toLowerCase() : "#fff";
            blinkingStatusDiv.style.color = cssColor === "yellow" ? "#000" : "#fff";
            blinkingStatusDiv.style.background =
                cssColor === "yellow" ? "rgba(255,255,0,0.8)" : "rgba(0,0,0,0.6)";

            blinkingStatusDiv.style.background = "rgba(0,0,0,0.6)";

            chossenSpeed = "uncertain";

        if (
            !rgHandled &&
            !failureUploadHandled &&
            !uploadInProgress &&
            rgDetector.lastMatchTime &&
            rgDetector.firstMatchTime &&
            rgDetector.lastMatchTime - rgDetector.firstMatchTime > 2000
        ) {
            rgHandled = true;
            processingStopped = true;

            scanEndTime = Date.now();
            scanDuration = scanEndTime - scanStartTime;

            console.log("RG Confirmed!, Notifying App");
            callLogInfo("RG confirmed. Uploading video once.");

            let ledStatus = ledDetector.getStatus();

            if (ledStatus) {
                choosenColor = ledDetector.lastDetectedColor;
                chossenSpeed = ledStatus.solid ? "solid" : (ledStatus.hz > .66 ? "fast" : "slow");
            } else {
                choosenColor = "off";
                chossenSpeed = "uncertain";
            }

            uploadCurrentVideo("RG_CONFIRMED", true);

            let msg = JSON.stringify({
                rgIdentified: true,
                rgModel: "BGW320",
                leds: {
                    serviceLed: {
                        color: choosenColor,
                        speed: chossenSpeed
                    },
                    wpsLed: {
                        color: "off",
                        speed: "uncertain"
                    }
                },
                debug: {
                    failureReason: "",
                    scanDuration: scanDuration
                }
            });

            setTimeout(() => {
                try {
                    window.ReactNativeWebView.postMessage(msg);
                } catch (e) {}

                try {
                    window.ThinNative.postMessage(msg);
                } catch (e) {}

                try {
                    window.webkit.messageHandlers.nativeCallBack.postMessage(msg);
                } catch (e) {}
            }, 300);

            frame.delete();
            return;
        }

        // Draw visual indicators on the output canvas only when viewing index.html
        const isIndexPage = window.location.href.includes("index.html");
        if (isIndexPage) {
            ledDetector.drawResult(results, outputCanvas);
            ledDetector.updateDebugDisplay();
        }

        // Clean up
        frame.delete();

        if (loopError) {
            document.getElementById("error-message").textContent = "";
            loopError = false;
        }

    } catch (err) {
        loopError = true;
        console.error("Error in main loop:", err);

        const el = document.getElementById("error-message");
        let stack = (err && err.stack) ? err.stack : "";

        if (!stack) {
            try {
                stack = new Error().stack || "";
            } catch (_) {}
        }

        let locationLine = "";

        if (stack) {
            const frames = stack.split("\n").map(l => l.trim());
            const target = frames.find(l => /index\.html|\.(js|ts|mjs|jsx|tsx)/i.test(l)) || frames[1] || frames[0];
            const m = target && target.match(/(?:\()?(.*?):(\d+):(\d+)\)?$/);

            if (m) {
                locationLine = `${m[1]}:${m[2]}:${m[3]}`;
            }
        }

        const msg = [
            `Loop Error: ${err && err.message ? err.message : String(err)}`,
            locationLine ? `Location: ${locationLine}` : '',
            stack ? `Stack trace:\n${stack}` : ''
        ].filter(Boolean).join('\n');

        console.error(msg);
        el.textContent = msg;
    }

    setTimeout(processFrame, 50);
}
