mediaRecorder.ondataavailable = function (event) {
    if (event.data && event.data.size > 0) {
        fullChunks.push(event.data);

        while (fullChunks.length > MAX_VIDEO_CHUNKS) {
            fullChunks.shift();
        }

        callLogInfo(
            'chunk added. chunks=' +
            fullChunks.length +
            ', chunkSize=' +
            event.data.size
        );
    }
};

mediaRecorder.start(RECORDER_TIMESLICE_MS);

function getFullVideoBlob() {
    if (!fullChunks.length) {
        callLogInfo('No video chunks available');
        return null;
    }

    const blob = new Blob(fullChunks, {
        type: 'video/webm'
    });

    callLogInfo(
        'Created latest video blob. chunks=' +
        fullChunks.length +
        ', size=' +
        blob.size
    );

    return blob;
}

function uploadCurrentVideo(reason, rgIdentifiedFlag) {
    try {
        if (uploadInProgress || processingStopped) {
            callLogInfo(
                'Upload skipped. uploadInProgress=' +
                uploadInProgress +
                ', processingStopped=' +
                processingStopped
            );
            return;
        }

        uploadInProgress = true;

        callLogInfo('uploadCurrentVideo started. reason=' + reason);

        const videoBlob = getFullVideoBlob();

        if (!videoBlob) {
            callLogInfo('uploadCurrentVideo stopped. No video blob.');
            uploadInProgress = false;
            return;
        }

        const ban = getQueryParam("ban") || "";

        const formData = new FormData();
        formData.append('video', videoBlob, 'last5seconds.webm');
        formData.append('ban', ban);
        formData.append('scanTime', scanDuration || (Date.now() - scanStartTime));
        formData.append('blinkSpeed', chossenSpeed || 'uncertain');
        formData.append('blinkColor', choosenColor || 'off');
        formData.append('rgIdentified', rgIdentifiedFlag ? 'true' : 'false');
        formData.append('failureReason', reason || '');

        callLogInfo(
            'calling uploadVideo. size=' +
            videoBlob.size +
            ', rgIdentified=' +
            rgIdentifiedFlag
        );

        fetch('/eapi/1/aiService/uploadVideo', {
            method: 'POST',
            body: formData
        })
        .then(res => {
            callLogInfo('uploadVideo response status=' + res.status);
            return res.text();
        })
        .then(t => {
            callLogInfo('uploadVideo success=' + t);
        })
        .catch(err => {
            callLogInfo('uploadVideo error=' + (err.message || err));
        })
        .finally(() => {
            uploadInProgress = false;
            processingStopped = true;
            callLogInfo('upload finished. processingStopped=true');
        });

    } catch (e) {
        uploadInProgress = false;
        callLogInfo('uploadCurrentVideo exception=' + (e.message || e));
    }
}

function processFrame() {
    if (processingStopped) {
        return;
    }

    try {
        if (!scanStartTime) {
            scanStartTime = Date.now();
        }

        if (
            !rgHandled &&
            !failureUploadHandled &&
            scanStartTime &&
            (Date.now() - scanStartTime > MAX_SCAN_TIME_MS)
        ) {
            failureUploadHandled = true;
            processingStopped = true;

            scanEndTime = Date.now();
            scanDuration = scanEndTime - scanStartTime;

            callLogInfo('RG not confirmed within timeout. Uploading failed scan video.');

            uploadCurrentVideo('RG_NOT_CONFIRMED_TIMEOUT', false);

            const msgObj = {
                rgIdentified: false,
                rgModel: "UNKNOWN",
                leds: {
                    serviceLed: {
                        color: choosenColor || "off",
                        speed: chossenSpeed || "uncertain"
                    },
                    wpsLed: {
                        color: "off",
                        speed: "uncertain"
                    }
                },
                debug: {
                    failureReason: "RG_NOT_CONFIRMED_TIMEOUT",
                    scanDuration: scanDuration
                }
            };

            setTimeout(() => {
                sendNativeMessage(msgObj);
            }, 300);

            return;
        }

        document.getElementById("cameraDimensions").textContent =
            `${tfTester.rgConfidence} || ${rgDetector.lastMean.toFixed(1)}`;

        if (preview.readyState !== 4) {
            setTimeout(processFrame, 50);
            return;
        }

        hiddenContext.drawImage(
            preview,
            0,
            0,
            hiddenCanvas.width,
            hiddenCanvas.height
        );

        (async () => {
            await tfTester.predict(hiddenCanvas);
        })();

        if (!tfTester.rgFound) {
            rgDetector.overlayImg.src = "ghost2.png";

            if (
                tfTester.lastSeen !== 0 &&
                (Date.now() - tfTester.lastSeen) > 1000
            ) {
                blinkingStatusDiv.textContent = "---";
                detectionResultDiv.textContent = "";
            }

            setTimeout(processFrame, 50);
            return;
        }

        let frame = cv.imread(hiddenCanvas);

        const results = ledDetector.detectLEDColor(frame);
        rgDetector.checkIfRg(frame);

        if (results.length > 0) {
            const r0 = results[0];

            choosenColor = r0.color;

            if (r0.isSolid) {
                chossenSpeed = "solid";
            } else if (r0.blinkHz && r0.blinkHz > 0) {
                chossenSpeed = r0.blinkHz > .66 ? "fast" : "slow";
            } else {
                chossenSpeed = "uncertain";
            }

            detectionResultDiv.textContent =
                `Detected: ${r0.color} (Hue: ${Math.round(r0.hue)})`;

            detectionResultDiv.style.color =
                r0.color.toLowerCase();

            blinkingStatusDiv.textContent =
                r0.isSolid
                    ? `${r0.color}: steady`
                    : `${r0.color}: blinking ${r0.blinkHz || 0} Hz`;

        } else if (Date.now() - ledDetector.detectionTimestamp > 1000) {
            choosenColor = "off";
            chossenSpeed = "uncertain";

            detectionResultDiv.textContent = "No LED detected";
            detectionResultDiv.style.color = "black";
            blinkingStatusDiv.textContent = "---";
        }

        if (
            !rgHandled &&
            rgDetector.lastMatchTime &&
            rgDetector.firstMatchTime &&
            rgDetector.lastMatchTime - rgDetector.firstMatchTime > 2000
        ) {
            rgHandled = true;
            processingStopped = true;

            scanEndTime = Date.now();
            scanDuration = scanEndTime - scanStartTime;

            callLogInfo('RG confirmed. Uploading latest 5 second scan video.');

            let ledStatus = ledDetector.getStatus();

            if (ledStatus) {
                choosenColor = ledDetector.lastDetectedColor;
                chossenSpeed = ledStatus.solid
                    ? "solid"
                    : (ledStatus.hz > .66 ? "fast" : "slow");
            } else {
                choosenColor = "off";
                chossenSpeed = "uncertain";
            }

            uploadCurrentVideo('RG_CONFIRMED', true);

            const msgObj = {
                rgIdentified: true,
                rgModel: "BGW320",
                leds: {
                    serviceLed: {
                        color: choosenColor || "off",
                        speed: chossenSpeed || "uncertain"
                    },
                    wpsLed: {
                        color: "off",
                        speed: "uncertain"
                    }
                },
                debug: {
                    failureReason: "",
                    scanDuration: scanDuration
                }
            };

            setTimeout(() => {
                sendNativeMessage(msgObj);
            }, 300);

            frame.delete();
            return;
        }

        const isIndexPage =
            window.location.href.includes('index.html');

        if (isIndexPage) {
            ledDetector.drawResult(results, outputCanvas);
            ledDetector.updateDebugDisplay();
        }

        frame.delete();

        if (loopError) {
            document.getElementById('error-message').textContent = '';
            loopError = false;
        }

    } catch (err) {
        loopError = true;

        callLogInfo('processFrame error=' + (err.message || err));

        console.error("Error in main loop:", err);

        const el = document.getElementById('error-message');

        if (el) {
            el.textContent =
                err && err.message ? err.message : String(err);
        }
    }

    setTimeout(processFrame, 50);
}
